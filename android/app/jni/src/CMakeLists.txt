# The game: every file of the Wii U game except its Wii U-only parts (src/platform: drawing, controllers, sound output),
# which are replaced by the files in android/port/.
cmake_minimum_required(VERSION 3.6)
set(PORT ${CMAKE_CURRENT_SOURCE_DIR}/../../../port)
file(GLOB GAME_SRC ${WIIU_SRC}/game/*.c ${WIIU_SRC}/audio/*.c ${WIIU_SRC}/game/*.cpp)
add_library(kkassets OBJECT ${WIIU_SRC}/render/assets.c)
target_compile_definitions(kkassets PRIVATE gfx_texture_rgba=gfx_texture_rgba_bytes gfx_texture_rgba_ex=gfx_texture_rgba_ex_bytes)
# Effekseer effects (phone only): the runtime and its OpenGL ES 3 renderer, built as a static library
set(EFK ${PORT}/effekseer)
file(GLOB_RECURSE EFK_SRC ${EFK}/*.cpp)
add_library(efk STATIC ${EFK_SRC} ${PORT}/efk.cpp)
target_compile_definitions(efk PRIVATE __EFFEKSEER_RENDERER_GLES3__ GL_GLEXT_PROTOTYPES KK_CONTENT_DIR="content/")
target_include_directories(efk PRIVATE ${EFK}/Effekseer ${EFK}/EffekseerRendererCommon ${EFK}/EffekseerRendererGL ${WIIU_SRC} ${PORT})
target_compile_options(efk PRIVATE -O2 -w -std=gnu++17 -include ${EFK}/efk_prefix.h)
add_library(main SHARED ${GAME_SRC} ${PORT}/gfx_gles.c ${PORT}/sdl_main.c ${PORT}/kk_files.c $<TARGET_OBJECTS:kkassets>)
set(KK_FLAGS KK_CONTENT_DIR="content/" _GNU_SOURCE)
target_compile_definitions(main PRIVATE ${KK_FLAGS})
target_compile_definitions(kkassets PRIVATE ${KK_FLAGS})
foreach(t main kkassets)
  target_include_directories(${t} PRIVATE ${WIIU_SRC} ${PORT} ${CMAKE_CURRENT_SOURCE_DIR}/../SDL/include)
  target_compile_options(${t} PRIVATE -O3 -w -ffast-math -fno-math-errno $<$<COMPILE_LANGUAGE:C>:-std=gnu11> $<$<COMPILE_LANGUAGE:C>:-include> $<$<COMPILE_LANGUAGE:C>:${PORT}/kk_files.h> $<$<COMPILE_LANGUAGE:CXX>:-std=gnu++17> $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions> $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>)
endforeach()
target_link_libraries(main efk SDL2 GLESv3 EGL android log m)
